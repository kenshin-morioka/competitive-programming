// 自分の回答 (TLE)
use proconio::input;

fn main() {
    input! {
        n:usize,
        q:usize,
        a:[u32;n],
        lr:[(usize, usize);q],
    }

    for i in lr {
        let tmp = &a[(i.0 - 1)..i.1];
        let ans: u32 = tmp.iter().sum();
        println!("{}", ans);
    }
}

// memo1:
// a[2]       // i32   → サイズが分かる → OK
// a[0..2]    // [i32] → サイズが分からない → そのまま値として扱えない
// &a[0..2]   // &[i32] → 参照自体のサイズは分かる → OK
// memo2:
// Rust のスライス範囲 start..end は 「start は含む、end は含まない」
