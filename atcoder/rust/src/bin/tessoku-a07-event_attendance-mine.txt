// 自分の回答 (ACだが O(N * D))
use proconio::input;
use std::io::{self, BufWriter, Write};

fn main() {
    input! {
        d:usize,
        n:usize,
        lr:[(usize,usize);n],
    }

    let mut ans = vec![0u64; d + 1];

    let out = io::stdout();
    let mut out = BufWriter::new(out.lock());

    for (l, r) in lr {
        for i in l..=r {
            ans[i] += 1;
        }
    }
    for i in 1..=d {
        writeln!(out, "{}", ans[i]).unwrap();
    }
}
